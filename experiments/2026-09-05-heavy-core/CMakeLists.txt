cmake_minimum_required(VERSION 3.20)
project(heavy_core_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(APXCHOL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(APXCHOL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(APXCHOL_USE_CUDA OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. apxchol)
include(../../cmake/apxchol_native_arch.cmake)
add_executable(heavy_core_bench driver.cpp)
target_link_libraries(heavy_core_bench PRIVATE apxchol_core apxchol_mtx_input fast_matrix_market::fast_matrix_market atomic)
target_compile_options(heavy_core_bench PRIVATE -Wall -Wextra -Wpedantic ${APXCHOL_NATIVE_ARCH_FLAG})
add_executable(heavy_core_check check.cpp)
target_link_libraries(heavy_core_check PRIVATE apxchol_core)
target_compile_options(heavy_core_check PRIVATE -Wall -Wextra -Wpedantic ${APXCHOL_NATIVE_ARCH_FLAG})
