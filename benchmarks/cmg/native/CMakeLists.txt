cmake_minimum_required(VERSION 3.18)
project(cmg_general_driver LANGUAGES CXX)
set(CMG_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmg-coder-arm64-cb23c3b/generated" CACHE PATH "Private recovered Coder source")
if(NOT EXISTS "${CMG_GENERATED_DIR}/cmg_pcg_packed.cpp")
  message(FATAL_ERROR "Set CMG_GENERATED_DIR to the private generated packed CMG source")
endif()
set(FMM_SOURCE_DIR "" CACHE PATH "Existing fast_matrix_market checkout")
if(NOT EXISTS "${FMM_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set FMM_SOURCE_DIR to the existing fast_matrix_market source")
endif()
set(EIGEN_INCLUDE_DIR "" CACHE PATH "Optional bundled Eigen include root")
if(EIGEN_INCLUDE_DIR)
  if(NOT EXISTS "${EIGEN_INCLUDE_DIR}/Eigen/Core")
    message(FATAL_ERROR "EIGEN_INCLUDE_DIR must contain Eigen/Core")
  endif()
  add_library(Eigen3::Eigen INTERFACE IMPORTED)
  set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${EIGEN_INCLUDE_DIR}")
else()
  find_package(Eigen3 3.3 REQUIRED NO_MODULE)
endif()
find_package(OpenMP REQUIRED COMPONENTS CXX)
# Use the maintained parser with its standard-library number conversion fallback.
set(FMM_USE_FAST_FLOAT OFF CACHE BOOL "" FORCE)
set(FMM_USE_DRAGONBOX OFF CACHE BOOL "" FORCE)
set(FMM_USE_RYU OFF CACHE BOOL "" FORCE)
add_subdirectory("${FMM_SOURCE_DIR}" fmm EXCLUDE_FROM_ALL)
file(GLOB CMG_SOURCES CONFIGURE_DEPENDS "${CMG_GENERATED_DIR}/*.cpp")
add_library(cmg_generated STATIC ${CMG_SOURCES})
target_include_directories(cmg_generated PUBLIC "${CMG_GENERATED_DIR}")
target_compile_features(cmg_generated PUBLIC cxx_std_17)
target_link_libraries(cmg_generated PUBLIC OpenMP::OpenMP_CXX)
# Redirect exactly the PCG translation unit's setup call through a timer.
# No generated file is modified and arithmetic/iteration control are unchanged.
set_source_files_properties("${CMG_GENERATED_DIR}/cmg_pcg_packed.cpp"
  PROPERTIES COMPILE_DEFINITIONS "cmg_setup_packed=cmg_setup_packed_timed")
add_executable(cmg_driver driver.cpp setup_timer.cpp)
target_link_libraries(cmg_driver PRIVATE cmg_generated Eigen3::Eigen fast_matrix_market::fast_matrix_market)
